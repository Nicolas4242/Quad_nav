import serial
import threading
import time

SERIAL_PORT = "/dev/ttyUSB0"
BAUD_RATE = 115200

def read_from_arduino(ser):
    while True:
        try:
            line = ser.readline().decode("utf-8", errors="ignore").strip()

            if line:
                print(f"\nArduino says: {line}")
                print("Command > ", end="", flush=True)

        except Exception as e:
            print(f"\nRead error: {e}")
            break

def main():
    try:
        ser = serial.Serial(SERIAL_PORT, BAUD_RATE, timeout=1)
    except Exception as e:
        print("Could not open serial port.")
        print(e)
        print("Check if CP2102 is connected and if the port is /dev/ttyUSB0.")
        return

    time.sleep(2)

    print("Connected to Arduino through CP2102.")
    print("Commands: start, stop, status, quit")

    reader_thread = threading.Thread(target=read_from_arduino, args=(ser,), daemon=True)
    reader_thread.start()

    while True:
        command = input("Command > ").strip()

        if command.lower() == "quit":
            print("Exiting.")
            break

        ser.write((command + "\n").encode("utf-8"))

    ser.close()

if __name__ == "__main__":
    main()
